            if (confirm("Adakah anda pasti ingin memuat semula data demo? Semua perubahan semasa akan digantikan.")) {
                assets = [...initialDemoAssets];
                saveAssets();
                renderTable();
                updateStats();
                showToast("Data demo berjaya dimuat semula!");
            }
        }

        // Render Table Rows
        function renderTable() {
            const tbody = document.getElementById('assetTableBody');
            const emptyState = document.getElementById('emptyState');
            const recordText = document.getElementById('recordCountText');

            const searchVal = document.getElementById('searchInput').value.toLowerCase().trim();
            const categoryVal = document.getElementById('categoryFilter').value;
            const statusVal = document.getElementById('statusFilter').value;

            // Filter logic
            const filteredAssets = assets.filter(asset => {
                const matchesSearch = asset.name.toLowerCase().includes(searchVal) ||
                                     asset.serialNo.toLowerCase().includes(searchVal) ||
                                     asset.location.toLowerCase().includes(searchVal);
                const matchesCategory = categoryVal === "" || asset.category === categoryVal;
                const matchesStatus = statusVal === "" || asset.status === statusVal;

                return matchesSearch && matchesCategory && matchesStatus;
            });

            tbody.innerHTML = '';

            if (filteredAssets.length === 0) {
                emptyState.classList.remove('hidden');
            } else {
                emptyState.classList.add('hidden');
                
                filteredAssets.forEach(asset => {
                    const tr = document.createElement('tr');
                    tr.className = "hover:bg-slate-50/80 transition text-slate-700 border-b border-slate-100";

                    // Status Badge Styling
                    let statusBadge = "";
                    if (asset.status === 'Baik') {
                        statusBadge = `<span class="px-2.5 py-1 text-xs font-semibold rounded-full bg-emerald-100 text-emerald-800 border border-emerald-200 flex items-center w-max"><span class="w-1.5 h-1.5 rounded-full bg-emerald-500 mr-1.5"></span>Baik</span>`;
                    } else if (asset.status === 'Penyelenggaraan') {
                        statusBadge = `<span class="px-2.5 py-1 text-xs font-semibold rounded-full bg-amber-100 text-amber-800 border border-amber-200 flex items-center w-max"><span class="w-1.5 h-1.5 rounded-full bg-amber-500 mr-1.5"></span>Penyelenggaraan</span>`;
                    } else {
                        statusBadge = `<span class="px-2.5 py-1 text-xs font-semibold rounded-full bg-rose-100 text-rose-800 border border-rose-200 flex items-center w-max"><span class="w-1.5 h-1.5 rounded-full bg-rose-500 mr-1.5"></span>Rosak</span>`;
                    }

                    tr.innerHTML = `
                        <td class="py-3 px-4 font-semibold text-slate-900">${escapeHtml(asset.name)}</td>
                        <td class="py-3 px-4 text-slate-600">${escapeHtml(asset.category)}</td>
                        <td class="py-3 px-4 font-mono text-xs text-slate-500">${escapeHtml(asset.serialNo)}</td>
                        <td class="py-3 px-4 text-slate-600">${escapeHtml(asset.location)}</td>
                        <td class="py-3 px-4 text-slate-500 text-xs">${asset.purchaseDate ? formatDate(asset.purchaseDate) : '-'}</td>
                        <td class="py-3 px-4 font-semibold text-slate-800">RM ${parseFloat(asset.price || 0).toLocaleString('ms-MY', {minimumFractionDigits: 2, maximumFractionDigits: 2})}</td>
                        <td class="py-3 px-4">${statusBadge}</td>
                        <td class="py-3 px-4 text-center no-print">
                            <div class="flex items-center justify-center space-x-2">
                                <button onclick="viewAsset('${asset.id}')" title="Lihat Butiran" class="p-1.5 text-slate-500 hover:text-brand-600 hover:bg-brand-50 rounded-lg transition">
                                    <i data-lucide="eye" class="w-4 h-4"></i>
                                </button>
                                <button onclick="openModal('edit', '${asset.id}')" title="Kemaskini" class="p-1.5 text-slate-500 hover:text-amber-600 hover:bg-amber-50 rounded-lg transition">
                                    <i data-lucide="edit" class="w-4 h-4"></i>
                                </button>
                                <button onclick="deleteAsset('${asset.id}')" title="Hapus Aset" class="p-1.5 text-slate-500 hover:text-rose-600 hover:bg-rose-50 rounded-lg transition">
                                    <i data-lucide="trash-2" class="w-4 h-4"></i>
                                </button>
                            </div>
                        </td>
                    `;
                    tbody.appendChild(tr);
                });
            }

            recordText.textContent = `Menunjukkan ${filteredAssets.length} daripada ${assets.length} rekod aset`;
            lucide.createIcons();
        }

        // Update Dashboard Cards
        function updateStats() {
            const totalCount = assets.length;
            const totalValue = assets.reduce((sum, item) => sum + (parseFloat(item.price) || 0), 0);
            const goodCount = assets.filter(a => a.status === 'Baik').length;
            const maintCount = assets.filter(a => a.status === 'Penyelenggaraan').length;
            const damagedCount = assets.filter(a => a.status === 'Rosak').length;

            document.getElementById('statTotalCount').textContent = totalCount;
            document.getElementById('statTotalValue').textContent = "RM " + totalValue.toLocaleString('ms-MY', {minimumFractionDigits: 2, maximumFractionDigits: 2});
            document.getElementById('statGoodCount').textContent = goodCount;
            document.getElementById('statMaintCount').textContent = maintCount;
            document.getElementById('statDamagedCount').textContent = damagedCount;
        }

        // Filter trigger
        function filterAssets() {
            renderTable();
        }

        function resetFilters() {
            document.getElementById('searchInput').value = '';
            document.getElementById('categoryFilter').value = '';
            document.getElementById('statusFilter').value = '';
            renderTable();
        }

        // Modal Operations
        function openModal(mode, assetId = null) {
            const modal = document.getElementById('assetModal');
            const form = document.getElementById('assetForm');
            const title = document.getElementById('modalTitle');

            form.reset();

            if (mode === 'edit' && assetId) {
                const asset = assets.find(a => a.id === assetId);
                if (asset) {
                    title.textContent = "Kemaskini Maklumat Aset";
                    document.getElementById('assetId').value = asset.id;
                    document.getElementById('assetName').value = asset.name;
                    document.getElementById('assetCategory').value = asset.category;
                    document.getElementById('assetSerial').value = asset.serialNo;
                    document.getElementById('assetLocation').value = asset.location;
                    document.getElementById('assetStatus').value = asset.status;
                    document.getElementById('assetPurchaseDate').value = asset.purchaseDate || '';
                    document.getElementById('assetPrice').value = asset.price;
                    document.getElementById('assetNotes').value = asset.notes || '';
                }
            } else {
                title.textContent = "Tambah Aset Baru";
                document.getElementById('assetId').value = '';
            }

            modal.classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('assetModal').classList.add('hidden');
        }

        function handleFormSubmit(e) {
            e.preventDefault();

            const id = document.getElementById('assetId').value;
            const name = document.getElementById('assetName').value.trim();
            const category = document.getElementById('assetCategory').value;
            const serialNo = document.getElementById('assetSerial').value.trim();
            const location = document.getElementById('assetLocation').value.trim();
            const status = document.getElementById('assetStatus').value;
            const purchaseDate = document.getElementById('assetPurchaseDate').value;
            const price = parseFloat(document.getElementById('assetPrice').value) || 0;
            const notes = document.getElementById('assetNotes').value.trim();

            if (id) {
                // Edit existing
                const index = assets.findIndex(a => a.id === id);
                if (index !== -1) {
                    assets[index] = { id, name, category, serialNo, location, status, purchaseDate, price, notes };
                    showToast("Maklumat aset berjaya dikemaskini!");
                }
            } else {
                // Add new
                const newAsset = {
                    id: "AST-" + Date.now().toString().slice(-5),
                    name,
                    category,
                    serialNo,
                    location,
                    status,
                    purchaseDate,
                    price,
                    notes
                };
                assets.unshift(newAsset);
                showToast("Aset baru berjaya ditambah!");
            }

            saveAssets();
            renderTable();
            updateStats();
            closeModal();
        }

        // View Asset Details
        function viewAsset(id) {
            const asset = assets.find(a => a.id === id);
            if (!asset) return;

            document.getElementById('detailName').textContent = asset.name;
            document.getElementById('detailCategory').textContent = asset.category;
            document.getElementById('detailSerial').textContent = asset.serialNo;
            document.getElementById('detailLocation').textContent = asset.location;
            document.getElementById('detailDate').textContent = asset.purchaseDate ? formatDate(asset.purchaseDate) : 'Tiada rekod';
            document.getElementById('detailPrice').textContent = "RM " + parseFloat(asset.price).toLocaleString('ms-MY', {minimumFractionDigits: 2});
            document.getElementById('detailNotes').textContent = asset.notes || 'Tiada catatan.';

            const detailBadge = document.getElementById('detailBadge');
            if (asset.status === 'Baik') {
                detailBadge.className = "px-2.5 py-1 text-xs font-semibold rounded-full bg-emerald-100 text-emerald-800 border border-emerald-200 inline-block";
            } else if (asset.status === 'Penyelenggaraan') {
                detailBadge.className = "px-2.5 py-1 text-xs font-semibold rounded-full bg-amber-100 text-amber-800 border border-amber-200 inline-block";
            } else {
                detailBadge.className = "px-2.5 py-1 text-xs font-semibold rounded-full bg-rose-100 text-rose-800 border border-rose-200 inline-block";
            }
            detailBadge.textContent = asset.status;

            document.getElementById('detailModal').classList.remove('hidden');
        }

        function closeDetailModal() {
            document.getElementById('detailModal').classList.add('hidden');
        }

        // Delete Asset
        function deleteAsset(id) {
            const asset = assets.find(a => a.id === id);
            if (!asset) return;

            if (confirm(`Adakah anda pasti ingin menghapus aset "${asset.name}" (${asset.serialNo})?`)) {
                assets = assets.filter(a => a.id !== id);
                saveAssets();
                renderTable();
                updateStats();
                showToast("Aset telah berjaya dihapuskan.");
            }
        }

        // Export to CSV
        function exportToCSV() {
            if (assets.length === 0) {
                alert("Tiada data aset untuk dieksport.");
                return;
            }

            const headers = ["ID", "Nama Aset", "Kategori", "No. Siri", "Lokasi", "Tarikh Beli", "Harga (RM)", "Status", "Catatan"];
            const rows = assets.map(a => [
                a.id,
                `"${a.name.replace(/"/g, '""')}"`,
                `"${a.category}"`,
                `"${a.serialNo}"`,
                `"${a.location}"`,
                a.purchaseDate || '',
                a.price,
                `"${a.status}"`,
                `"${(a.notes || '').replace(/"/g, '""')}"`
            ]);

            const csvContent = "data:text/csv;charset=utf-8,\uFEFF" 
                + [headers.join(","), ...rows.map(e => e.join(","))].join("\n");

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `Laporan_Aset_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);

            showToast("Fail CSV berjaya dimuat turun!");
        }

        // Helper: Format Date
        function formatDate(dateStr) {
            if (!dateStr) return '';
            const d = new Date(dateStr);
            return d.toLocaleDateString('ms-MY', { day: '2-digit', month: '2-digit', year: 'numeric' });
        }

        // Helper: Escape HTML string
        function escapeHtml(str) {
            return str ? str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;") : '';
        }

        // Toast Notification
        function showToast(message) {
            const toast = document.getElementById('toast');
            const toastMessage = document.getElementById('toastMessage');
            const toastIcon = document.getElementById('toastIcon');

            toastMessage.textContent = message;
            toastIcon.innerHTML = `<i data-lucide="check-circle-2" class="w-5 h-5 text-emerald-400"></i>`;
            lucide.createIcons();

            toast.classList.remove('translate-y-20', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');

            setTimeout(() => {
                toast.classList.remove('translate-y-0', 'opacity-100');
                toast.classList.add('translate-y-20', 'opacity-0');
            }, 3000);
        }
